# XG2010G-only 合并实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 `ImmortalWrt-for-Gemtek-brightspeed` 当前最新树导入 `xg2010g-immortalwrt-for-gemtek-brightspeed`，精简为只构建 Gemtek XG2010G 的仓库，同时保留后续同步 brightspeed 更新的能力。

**Architecture:** 用 `git read-tree -u --reset` 将 brightspeed 快照作为独立 commit 导入（不导入其提交历史），在其上提交 XG2010G-only 精简；用可重复应用的 trim patch 和 `check-xg2010g-only.sh` 维持约束；后续同步用 `git merge-tree --merge-base=<上次快照>` 做三点合并。

**Tech Stack:** OpenWrt/ImmortalWrt buildroot（Linux 6.18）、git 2.54、bash、GitHub Actions。

## Global Constraints

- 只保留 airoha target 的两个 XG2010G profile：`gemtek_xg2010g`（chainloader 厂商布局）和 `gemtek_xg2010g-ubi`（OpenWrt U-Boot UBI 布局）。
- 不导入 brightspeed 提交历史；导入 commit 的树必须与 brightspeed 快照完全一致。
- 不删除其他 target/架构、上游共享包、共享 SoC 驱动和内核补丁。
- 保留 `an7583` / `en7523` 子目标目录、`config-6.18` 和 `target.mk`，但不保留其设备 profile。
- 所有工作在本地分支 `xg2010g-only` 上完成，推送前必须得到用户确认。
- macOS 不能构建 OpenWrt；构建验证通过 GitHub Actions 完成。

define Device/gemtek_xg2010g-common
  DEVICE_VENDOR := Gemtek
  DEVICE_PACKAGES := fitblk kmod-leds-gpio kmod-gpio-button-hotplug \
	kmod-phy-airoha-en8811h kmod-phy-realtek rtl826x-firmware \
	kmod-nf-conntrack-bridge kmod-nft-bridge \
	kmod-airoha-en7572 kmod-airoha-xpon airoha-ponctl airoha-pond \

	-kmod-mac80211 -kmod-mt7996-firmware -kmod-mt7996e \
	-wpad-mbedtls -wpad-mesh-mbedtls -wireless-regdb
  BLOCKSIZE := 128k
  PAGESIZE := 2048
  UBINIZE_OPTS := -E 5
  KERNEL_IN_UBI := 1
  KERNEL := kernel-bin | gzip
  IMAGES := sysupgrade.itb
  # The dynamic UBI fit volume is recreated to the uploaded image size.
  # The 128 MiB package limit is independent of UBI volume placement.
  IMAGE_SIZE := 131072k
  IMAGE/sysupgrade.itb := append-kernel | \
	fit gzip $$(KDIR)/image-$$(firstword $$(DEVICE_DTS)).dtb external-static-with-rootfs | \
	append-metadata | check-size
  SOC := an7581
endef

define Device/gemtek_xg2010g-ubi-common
  DEVICE_VARIANT := UBI
  DEVICE_COMPAT_VERSION := 2.0
  DEVICE_COMPAT_MESSAGE := Firmware requires the XG2010G OpenWrt U-Boot UBI layout \
       with bl2 at 0x00000000 and ubi starting at 0x00020000, containing \
       fip, ubootenv, ubootenv2, factory and fit. Do not force an upgrade from \
       the legacy 0x00600000 layout. Preserve bl2 and the boot/calibration volumes.
endef

define Device/gemtek_xg2010g
  # DEVICE_DTS must be set before the common block: the common IMAGE line
  # uses := and freezes $(firstword $(DEVICE_DTS)) at eval time.
  DEVICE_DTS := an7581-xg2010g
  $(call Device/gemtek_xg2010g-common)
  DEVICE_MODEL := XG2010G
  SUPPORTED_DEVICES := gemtek,xg2010g
endef
TARGET_DEVICES += gemtek_xg2010g

define Device/gemtek_xg2010g-ubi

  # uses := and freezes $(firstword $(DEVICE_DTS)) at eval time.
  DEVICE_DTS := an7581-gemtek-xg2010g-ubi
  $(call Device/gemtek_xg2010g-common)
  $(call Device/gemtek_xg2010g-ubi-common)
  DEVICE_MODEL := XG2010G
  # One image covers the 1 GiB and 2 GiB boards: U-Boot publishes the DRAM
  # size detected by BL2, which overrides the fallback in the device tree.
  # The 2 GiB compatibles are kept so upgrades from the old variant work.
  SUPPORTED_DEVICES := gemtek,xg2010g-ubi gemtek,xg2010g \
	gemtek,xg2010g-2g-ubi gemtek,xg2010g-2g
endef
TARGET_DEVICES += gemtek_xg2010g-ubi
EOF

- [ ] **Step 5: 替换 platform.sh**

```bash
cat > target/linux/airoha/an7581/base-files/lib/upgrade/platform.sh <<'EOF'
RAMFS_COPY_BIN='fitblk fit_check_sign'

REQUIRE_IMAGE_METADATA=1

. /lib/upgrade/gemtek-ubi.sh

airoha_require_ubi_layout()
{
	local ubidev volume fip_volume

	ubidev="$(nand_find_ubi ubi)"
	[ -n "$ubidev" ] || {
		echo "UBI is not initialized; create the base volumes from U-Boot recovery."
		return 1
	}

	for volume in fip ubootenv ubootenv2 "$@"; do
		if ! nand_find_volume "$ubidev" "$volume" >/dev/null; then
			echo "UBI base volume $volume is missing; repair the layout from U-Boot recovery."
			return 1
		fi
	done

	fip_volume="$(nand_find_volume "$ubidev" fip)"
	[ "$(cat "/sys/class/ubi/$fip_volume/data_bytes")" -gt 0 ] || {
		echo "The fip volume does not contain BL31/U-Boot; install the boot chain first."
		return 1
	}
}

platform_check_image() {
	local board=$(board_name)

	[ "$#" -gt 1 ] && return 1

	case "$board" in
	gemtek,xg2010g-ubi|\
	gemtek,xg2010g-2g-ubi|\
	gemtek,xg2010g-2g)
		gemtek_ubi_layout_check "$board" || {
			echo "Unsupported Gemtek UBI layout; refusing sysupgrade."
			return 1
		}
		fit_check_image "$1"
		return $?
		;;
	gemtek,xg2010g)
		fit_check_image "$1"
		return $?
		;;
	esac

	return 0
}

platform_do_upgrade() {
	local board=$(board_name)

	case "$board" in
		gemtek,xg2010g-ubi|\
		gemtek,xg2010g-2g-ubi|\
		gemtek,xg2010g-2g)
			gemtek_ubi_layout_check "$board" &&
				airoha_require_ubi_layout factory && fit_do_upgrade "$1"
			;;
		gemtek,xg2010g)
			fit_do_upgrade "$1"
			;;
		*)
			nand_do_upgrade "$1"
			;;
	esac
}
EOF
```

```bash
cat > scripts/check-xg2010g-only.sh <<'EOF'
#!/usr/bin/env bash
set -Eeuo pipefail

cd "$(dirname "$0")/.."

fail=0
err() { echo "check-xg2010g-only: $*" >&2; fail=1; }

expected_devices=$'gemtek_xg2010g\ngemtek_xg2010g-ubi'
actual_devices=$(grep -h '^TARGET_DEVICES += ' target/linux/airoha/image/an7581.mk \
	| sed 's/^TARGET_DEVICES += //' | sort || true)
if [[ "$actual_devices" != "$expected_devices" ]]; then
	err "an7581 TARGET_DEVICES mismatch"
	diff -u <(printf '%s\n' "$expected_devices") <(printf '%s\n' "$actual_devices") >&2 || true
fi

if grep -H '^TARGET_DEVICES += ' target/linux/airoha/image/an7583.mk \
	target/linux/airoha/image/en7523.mk 2>/dev/null; then
	err "an7583/en7523 still define target devices"
fi

allowed_dts='an7581-gemtek-xg2010g-ubi.dts an7581-xg2010g.dts an7581.dtsi'
for f in target/linux/airoha/dts/*; do
	base=$(basename "$f")
	case " $allowed_dts " in
		*" $base "*) ;;
		*) err "unexpected DTS file: $f" ;;
	esac
done

for f in *.config; do
	case "$f" in
		2010.config) ;;
		*) err "unexpected build config: $f" ;;
	esac
done

selected=$(grep -E '^CONFIG_TARGET_airoha_an7581_DEVICE_.*=y$' 2010.config \
	| sed 's/.*_DEVICE_//; s/=y//' | sort || true)
case "$selected" in
	gemtek_xg2010g|gemtek_xg2010g-ubi) ;;
	*) err "2010.config must select exactly one XG2010G profile, got: $selected" ;;
esac

for path in \
	package/boot/uboot-airoha \
	target/linux/airoha/an7583/base-files \
	package/boot/uboot-tools/uboot-envtools/files/airoha_an7583 \
	1710.config 040.config 040-tf.config config.seed \
	.github/workflows/sync-upstream.yml; do
	[[ ! -e "$path" ]] || err "obsolete path still exists: $path"
done

if grep -Eq '^src-git .*(easytier|ac_agent)' feeds.conf.default; then
	err "feeds.conf.default still contains easytier/ac_agent"
fi
if grep -qx 'CONFIG_FEED_ac_agent=y' 2010.config; then
	err "2010.config still enables the ac_agent feed"
fi

if grep -Eq '1710\.config|040(-tf)?\.config|config_seed' .github/workflows/build-firmware.yml; then
	err "build workflow still references other configs"
fi
grep -q '2010\.config' .github/workflows/build-firmware.yml \
	|| err "build workflow does not use 2010.config"

if grep -RInE 'Device/(airoha_an7581-evb|gemtek_w1700k-ubi|nokia_|gemtek_xr1710g|quantum_q1000k)|TARGET_DEVICES \+= (airoha_an7581-evb|gemtek_w1700k-ubi|nokia_|gemtek_xr1710g|quantum_q1000k)' \
	target/linux/airoha/image/; then
	err "non-XG2010G device definitions remain"
fi

if grep -RInE 'gemtek,(xr1710g|w1700k)|nokia,|quantum,|fiberhome,|znxt,|unionman,' \
	target/linux/airoha/base-files target/linux/airoha/an7581/base-files; then
	err "non-XG2010G board branches remain"
fi

(( fail == 0 )) || exit 1
echo "check-xg2010g-only: ok"
EOF
chmod +x scripts/check-xg2010g-only.sh
bash scripts/check-xg2010g-only.sh
```

# ImmortalWrt for Gemtek XG2010G

基于 [ImmortalWrt](https://github.com/immortalwrt/immortalwrt) 的 Gemtek XG2010G 固件仓库。

cat > scripts/sync-brightspeed.sh <<'EOF'
#!/usr/bin/env bash
set -Eeuo pipefail

cd "$(dirname "$0")/.."

die() { echo "sync-brightspeed: $*" >&2; exit 1; }

[[ -f .brightspeed-upstream ]] || die "missing .brightspeed-upstream"
base=$(sed -n 's/^commit=//p' .brightspeed-upstream)
source_url=$(sed -n 's/^source=//p' .brightspeed-upstream)
[[ -n "$base" && -n "$source_url" ]] || die "invalid .brightspeed-upstream"

git diff --quiet && git diff --cached --quiet || die "worktree has uncommitted changes"

git fetch --no-tags "$source_url" master
new=$(git rev-parse FETCH_HEAD)
if [[ "$new" == "$base" ]]; then
	echo "already up to date: $new"
	exit 0
fi

echo "merging brightspeed ${base:0:8} -> ${new:0:8}"
out=$(mktemp)
trap 'rm -f "$out"' EXIT
if ! git merge-tree --write-tree --name-only --merge-base="$base" HEAD "$new" >"$out"; then
	echo "merge conflicts:" >&2
	tail -n +2 "$out" >&2
	exit 1
fi
tree=$(head -n 1 "$out")

branch="sync/brightspeed-$(date +%Y%m%d-%H%M%S)"
git switch -c "$branch"
git read-tree -u --reset "$tree"
git checkout-index -a -f

bash scripts/apply-xg2010g-trim.sh

cat > .brightspeed-upstream <<CONF
commit=$new
date=$(TZ=Asia/Shanghai date +%Y-%m-%d)
source=$source_url
CONF

git add -A
git commit -m "sync: brightspeed $(git rev-parse --short=8 "$new")"
echo "sync committed on $branch; review and push manually"
EOF
chmod +x scripts/sync-brightspeed.sh
bash -n scripts/apply-xg2010g-trim.sh
bash -n scripts/sync-brightspeed.sh
